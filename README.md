-- ==========================================
-- DATABASE ABSENSI SISWA
-- ==========================================

-- 1. Membuat tabel absensi
create table if not exists public.absensi (
  id bigint generated by default as identity primary key,
  nama text not null,
  status text not null check (status in ('Hadir', 'Tidak Hadir')),
  tanggal text not null,
  waktu text not null,
  zona text not null default 'WIB (UTC+7)',
  created_at timestamptz not null default now()
);


-- 2. Mengaktifkan Row Level Security
alter table public.absensi enable row level security;


-- 3. Mengizinkan siapa saja membaca data absensi
create policy "Absensi dapat dilihat semua orang"
on public.absensi
for select
to anon
using (true);


-- 4. Mengizinkan siswa mengirim absensi
create policy "Siswa dapat mengirim absensi"
on public.absensi
for insert
to anon
with check (true);


-- 5. Mengizinkan realtime
alter publication supabase_realtime
add table public.absensi;
